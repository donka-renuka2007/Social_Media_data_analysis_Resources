script-3:
-- Extract hour and assign time category using BINCOND
categorized = FOREACH posts GENERATE 
    post_id,
    polarity_score,
    (
        GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) >= 5 AND GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) <= 11 ? 'Morning' :
        (GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) >= 12 AND GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) <= 16 ? 'Afternoon' :
        (GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) >= 17 AND GetHour(ToDate(timestamp, 'yyyy-MM-dd HH:mm:ss')) <= 20 ? 'Evening' : 'Night'))
    ) AS time_category;

-- Group by time category
grouped_time = GROUP categorized BY time_category;

-- Calculate total post count and average sentiment polarity per category
analytics = FOREACH grouped_time GENERATE 
    group AS time_category, 
    COUNT(categorized.post_id) AS total_posts,
    AVG(categorized.polarity_score) AS avg_polarity;

-- Display result
DUMP analytics;
